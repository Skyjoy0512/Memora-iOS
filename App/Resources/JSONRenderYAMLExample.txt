YAMLの表示検証です。架空のデータを使います。

```yaml-spec
root: card
state:
  name: ""
  show: true
  tab: overview
  items:
    - id: a
      name: 検証用A
      done: false
    - id: b
      name: 検証用B
      done: true
  open: false
elements:
  card:
    type: Card
    props:
      title: 操作できる回答
      description: 架空の検証用データです。AIや外部サービスには送信しません。
    children:
      - name
      - greeting
      - switch
      - tabs
      - panel
      - repeat
      - dialogButton
      - dialog
      - copy
  name:
    type: Input
    props:
      label: 表示する名前
      name: name
      value:
        $bindState: /name
      placeholder: 名前を入力
      checks:
        - type: minLength
          args:
            min: 2
          message: 2文字以上で入力してください。
      validateOn: blur
    children: []
  greeting:
    type: Text
    props:
      text:
        $template: こんにちは、${/name}さん
    children: []
  switch:
    type: Switch
    props:
      label: 補足を表示
      name: show
      checked:
        $bindState: /show
    children: []
  tabs:
    type: Tabs
    props:
      tabs:
        - label: 概要
          value: overview
        - label: 詳細
          value: detail
      value:
        $bindState: /tab
    children:
      - overview
      - detail
  overview:
    type: Metric
    props:
      label: 検証用の項目数
      value:
        $count:
          $state: /items
      detail: 状態から自動で計算
    visible:
      $state: /tab
      eq: overview
    children: []
  detail:
    type: Text
    props:
      text: 詳細タブの表示内容です。
    visible:
      $state: /tab
      eq: detail
    children: []
  panel:
    type: Panel
    props:
      label: 補足
    visible:
      $state: /show
      eq: true
    children:
      - body
    slots:
      header:
        - header
      footer:
        - footer
  header:
    type: Badge
    props:
      text: 名前付きスロット
    children: []
  body:
    type: Text
    props:
      text: スイッチでこの補足の表示が切り替わります。
    children: []
  footer:
    type: Text
    props:
      text: この表示はローカルの状態です。
      variant: caption
    children: []
  repeat:
    type: Stack
    props:
      direction: vertical
      gap: sm
    repeat:
      statePath: /items
      key: id
    children:
      - item
  item:
    type: Checkbox
    props:
      label:
        $item: name
      name:
        $item: id
      checked:
        $bindItem: done
    children: []
  dialogButton:
    type: Button
    props:
      label: ダイアログを開く
      variant: secondary
    on:
      press:
        action: setState
        params:
          statePath: /open
          value: true
    children: []
  dialog:
    type: Dialog
    props:
      title: 確認用ダイアログ
      description: 閉じても入力内容は残ります。
      openPath: /open
    children:
      - dialogText
  dialogText:
    type: Text
    props:
      text: 本家のDialogコンポーネントです。
    children: []
  copy:
    type: Button
    props:
      label: 入力した名前をコピー
    on:
      press:
        action: copyText
        params:
          text:
            $state: /name
        onSuccess:
          set:
            /copied: true
    children: []

```
